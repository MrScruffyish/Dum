# Dum difficulty levels. The host (or dedicated server) picks one; it applies to everyone
# and is stored in the save. Shown in the main menu in this order. Add your own!
#
# monster_damage / monster_hp / monster_speed: multipliers
# monster_tohit: added to monster attack rolls
# alertness:     how quickly sleeping monsters notice you (multiplier)
# extra_groups:  more (or fewer, negative) monster groups per level
# hunger / fuel: how fast food and light run out (multipliers)
# respawn:       seconds until a dead player gets back up (while someone lives)
# party_wipe:    restart = the whole party dead starts a new game; revive = everyone gets up
# xp / gold:     multipliers for experience and gold found
# monster_count: multiplier for the number of monster groups (on top of extra_groups)
# monster_depth: added to the dungeon level when picking monsters. +1 makes the kinds that
#                normally wait until one level deeper appear now; -1 holds them back.
# danger:        multiplier for traps and dangers placed on a level: locked doors, vaults,
#                lever vaults, trapdoors and static floor hazards (1 = the normal amount)
# per_level:     each dungeon level below the first adds this much (0.5 = +50%) of the base
#                monster health, damage, xp and gold; monster to-hit +1 every 2 levels
# count_per_level: the same for the number of monsters (default: per_level)
# hidden = yes:  not shown in the difficulty menu (used by The Abyss)

# The Abyss (Adventure: The Abyss): an endless mode that keeps a high-score list. It starts
# like Easy, and every level down it keeps growing without bound: per_level adds 50% (of
# level 1's) to monster health, damage, xp and gold, count_per_level adds 20% more monsters,
# and the traps grow with them. Monsters appear at their normal depth (monster_depth = 0), so
# the deeper kinds arrive on the level they are meant to. A wipe sends the party two levels
# above its deepest descent (never below level 1) and takes part of its gold and experience;
# the deepest level reached goes on the high-score list.
[difficulty abyss]
name = The Abyss
desc = Endless: every level down, monsters get 50% stronger and 20% more numerous, forever.

monster_speed = 0.9

monster_depth = 0
